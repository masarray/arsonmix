cmake_minimum_required(VERSION 3.20)
project(DeEzzerDSP LANGUAGES CXX)
include(CTest)
option(DE_EZZER_SANITIZERS "ASan/UBSan" OFF)
add_executable(de_ezzer_test test.cpp)
target_compile_features(de_ezzer_test PRIVATE cxx_std_20)
set_target_properties(de_ezzer_test PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(de_ezzer_test PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(de_ezzer_test PRIVATE -Wall -Wextra -Werror)
endif()
if(DE_EZZER_SANITIZERS)
  target_compile_options(de_ezzer_test PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(de_ezzer_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME DeEzzerFoundation COMMAND de_ezzer_test)
option(DE_EZZER_BENCHMARK "Build callback timing probe" OFF)
if(DE_EZZER_BENCHMARK)
  add_executable(de_ezzer_bench bench.cpp)
  target_compile_features(de_ezzer_bench PRIVATE cxx_std_20)
  set_target_properties(de_ezzer_bench PROPERTIES CXX_EXTENSIONS OFF)
endif()
