cmake_minimum_required(VERSION 3.20)
project(AntiNoizDSP LANGUAGES CXX)
include(CTest)
option(ANTI_NOIZ_BENCHMARK "Build standalone throughput benchmark" OFF)
option(ANTI_NOIZ_SANITIZERS "Enable ASan and UBSan on GCC/Clang" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(anti_noiz_target name source)
  add_executable(${name} "${ROOT}/${source}")
  target_compile_features(${name} PRIVATE cxx_std_20)
  set_target_properties(${name} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
  endif()
  if(ANTI_NOIZ_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
      message(FATAL_ERROR "Sanitizers require GCC or Clang with GNU-style flags")
    endif()
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
anti_noiz_target(anti_noiz_core_test tests/anti_noiz_core_test.cpp)
add_test(NAME AntiNoizCore COMMAND anti_noiz_core_test)
set_tests_properties(AntiNoizCore PROPERTIES TIMEOUT 60)
if(ANTI_NOIZ_BENCHMARK)
  anti_noiz_target(anti_noiz_bench tests/anti_noiz_throughput_bench.cpp)
endif()
anti_noiz_target(anti_noiz_control_test tests/anti_noiz_control_test.cpp)
add_test(NAME AntiNoizControl COMMAND anti_noiz_control_test)
set_tests_properties(AntiNoizControl PROPERTIES TIMEOUT 60)
anti_noiz_target(anti_noiz_render tools/anti-noiz/offline_renderer.cpp)
anti_noiz_target(anti_noiz_offline_test tests/anti_noiz_offline_test.cpp)
add_test(NAME AntiNoizOffline COMMAND anti_noiz_offline_test)
set_tests_properties(AntiNoizOffline PROPERTIES TIMEOUT 60)
anti_noiz_target(anti_noiz_context_test tests/anti_noiz_context_test.cpp)
add_test(NAME AntiNoizContext COMMAND anti_noiz_context_test)
set_tests_properties(AntiNoizContext PROPERTIES TIMEOUT 60)
